local function makeSlider(name, caption, y, minVal, maxVal, initial, callback)
    local titleLabel = label(name .. "Label", caption, UDim2.fromOffset(14, y), UDim2.new(1, -28, 0, 18), panel)
    local valueLabel = label(name .. "Value", tostring(initial), UDim2.new(1, -70, 0, y), UDim2.fromOffset(56, 18), panel)
    valueLabel.TextXAlignment = Enum.TextXAlignment.Right
    valueLabel.TextColor3 = Color3.fromRGB(195, 195, 205)
    local track = Instance.new("Frame")
    track.Name = name .. "Track"
    track.Position = UDim2.fromOffset(14, y + 20)
    track.Size = UDim2.new(1, -28, 0, 8)
    track.BackgroundColor3 = Color3.fromRGB(54, 54, 62)
    track.BorderSizePixel = 0
    track.Active = true
    track.Parent = panel
    Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)
    local fill = Instance.new("Frame")
    fill.Name = "Fill"
    fill.Size = UDim2.new((initial - minVal) / (maxVal - minVal), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(230, 230, 238)
    fill.BorderSizePixel = 0
    fill.Parent = track
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)
    local knob = Instance.new("Frame")
    knob.Size = UDim2.fromOffset(14, 14)
    knob.AnchorPoint = Vector2.new(0.5, 0.5)
    knob.Position = UDim2.new((initial - minVal) / (maxVal - minVal), 0, 0.5, 0)
    knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knob.BorderSizePixel = 0
    knob.Parent = track
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)
    local dragging = false
    local value = initial
    local function setFromX(x)
        local alpha = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        value = minVal + alpha * (maxVal - minVal)
        local rounded = math.floor(value * 100 + 0.5) / 100
        fill.Size = UDim2.new(alpha, 0, 1, 0)
        knob.Position = UDim2.new(alpha, 0, 0.5, 0)
        valueLabel.Text = tostring(rounded)
        callback(value)
    end
    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    return titleLabel
end

local function ensureEffect(className, name)
    if post[name] then return post[name] end
    local existing = Lighting:FindFirstChild(name)
    if existing and existing.ClassName == className then
        post[name] = existing
    else
        local obj = Instance.new(className)
        obj.Name = name
        obj.Enabled = false
        obj.Parent = Lighting
        post[name] = obj
    end
    return post[name]
end

local function updateFovLabel()
    fovLabel.Text = "FOV: " .. tostring(math.floor(fov + 0.5)) .. "°"
end

local function tweenFov(target, duration)
    target = math.clamp(target, 30, 100)
    fov = target
    updateFovLabel()
    local cam = camera()
    if not cam then return end
    if tweens.fov then tweens.fov:Cancel() end
    tweens.fov = TweenService:Create(cam, TweenInfo.new(duration or 0.25, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {FieldOfView = target})
    tweens.fov:Play()
end

local function moveToSaved(cf)
    local cam = camera()
    if not cam then return end
    cam.CameraType = Enum.CameraType.Scriptable
    if tweens.move then tweens.move:Cancel() end
    tweens.move = TweenService:Create(cam, TweenInfo.new(transitionSeconds, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {CFrame = cf})
    tweens.move:Play()
    statusText("Smooth camera transition...")
    tweens.move.Completed:Once(function(state)
        if state == Enum.PlaybackState.Completed then statusText("Camera reached preset") end
    end)
end

local function captureFrame(now)
    local cam = camera()
    if not cam then return end
    table.insert(recordedFrames, {t = now - recordingStarted, cf = cam.CFrame, fov = cam.FieldOfView})
end

local function startRecording()
    if recording then return end
    recordedFrames = {}
    recording = true

local function stopRecording()
    if not recording then return end

end

local function playFrames(frames)
    if #frames < 2 then statusText("Record a move or save at least two camera positions first"); return end
    cutsceneToken += 1
    local myToken = cutsceneToken
    cutscenePlaying, cutscenePaused = true, false
    local cam = camera()
    if not cam then return end
    cutsceneReturnType, cutsceneReturnSubject = cam.CameraType, cam.CameraSubject
    cam.CameraType = Enum.CameraType.Scriptable
    fadeTo(0, 0.01)
    fadeTo(1, 0.6)
    statusText("Cutscene playing • " .. #frames .. " keyframes")
    task.spawn(function()
        local index = 1
        while index < #frames and cutsceneToken == myToken do
            if cutscenePaused then
                RunService.Heartbeat:Wait()
            else
                local a, b = frames[index], frames[index + 1]
                local duration = math.max(0.05, b.t - a.t)
                cam.CFrame = a.cf
                cam.FieldOfView = a.fov or cam.FieldOfView
                local tw = TweenService:Create(cam, TweenInfo.new(duration, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {
                    CFrame = b.cf,
                    FieldOfView = b.fov or cam.FieldOfView
                })
                tw:Play()
                local completed = false
                local connection
                connection = tw.Completed:Connect(function() completed = true end)
                while not completed and cutsceneToken == myToken do
                    if cutscenePaused then
                        tw:Pause()
                        repeat RunService.Heartbeat:Wait() until not cutscenePaused or cutsceneToken ~= myToken
                        if cutsceneToken == myToken then tw:Play() end
                    end
                    RunService.Heartbeat:Wait()
                end
                connection:Disconnect()
                if cutsceneToken ~= myToken then tw:Cancel(); return end
                index += 1
            end
        end
        if cutsceneToken == myToken then
            cutscenePlaying, cutscenePaused = false, false
            fadeTo(0, 0.25)
            task.wait(0.25)
            fadeTo(1, 0.5)
            if cam and cam.Parent then
                cam.CameraType = cutsceneReturnType or Enum.CameraType.Custom
                if cutsceneReturnSubject then cam.CameraSubject = cutsceneReturnSubject end
            end
            statusText("Cutscene finished")
        end
    end)
end

local function makeSavedPath()
    local frames = {}
    local timeCursor = 0
    for i = 1, 4 do
        if saved[i] then
            table.insert(frames, {t = timeCursor, cf = saved[i], fov = fov})
            timeCursor += transitionSeconds
        end
    end
    if #frames >= 2 then
        for i, frame in ipairs(frames) do frame.t = (i - 1) * transitionSeconds end
    end
    return frames
end

local function runCommand(raw)
    if typeof(raw) ~= "string" then return end
    local msg = string.lower(raw):match("^%s*(.-)%s*$")
    local n = tonumber(msg:match("^/fov%s+(%d+%.?%d*)$"))
    local t = tonumber(msg:match("^/camtime%s+(%d+%.?%d*)$"))
    local saveN = tonumber(msg:match("^/savepos%s+(%d+)$"))
    local goN = tonumber(msg:match("^/gotopos%s+(%d+)$"))
    if msg == "/hideeffects" then hidden = true; panel.Visible = false
    elseif msg == "/showeffects" then hidden = false; panel.Visible = true
    elseif msg == "/shake" then shakeButton:Activate()
    elseif msg == "/cinezoom" then cine:Activate()
    elseif msg == "/bloom" then bloomButton:Activate()
    elseif msg == "/blur" then blurButton:Activate()
    elseif msg == "/sunrays" then raysButton:Activate()
    elseif msg == "/depth" then dofButton:Activate()
    elseif msg == "/letterbox" then boxButton:Activate()
    elseif msg == "/monochrome" then monoButton:Activate()
    elseif msg == "/warm" then warmButton:Activate()
    elseif msg == "/resetcamfx" then resetBtn:Activate()
    elseif msg == "/recordcutscene" then startRecording()
    elseif msg == "/stoprecord" then stopRecording()
    elseif msg == "/playcutscene" then
        if recording then stopRecording() end
        if #recordedFrames >= 2 then playFrames(recordedFrames) else playFrames(makeSavedPath()) end
    elseif msg == "/pausecutscene" then
        if cutscenePlaying then
            cutscenePaused = not cutscenePaused
            pauseCutsceneBtn.Text = cutscenePaused and "▶ RESUME" or "Ⅱ PAUSE"
        end
    elseif msg == "/stopcutscene" then
        cutsceneToken += 1
        cutscenePlaying, cutscenePaused = false, false
        stopRecording()
        local cam = camera()
        if cam then
            cam.CameraType = cutsceneReturnType or Enum.CameraType.Custom
            if cutsceneReturnSubject then cam.CameraSubject = cutsceneReturnSubject end
        end
        fadeTo(1, 0.25)
    elseif n then tweenFov(n, 0.3)
    elseif t then transitionSeconds = math.clamp(t, 0.25, 15); statusText("Transition duration: " .. transitionSeconds .. "s")
    elseif saveN and saveN >= 1 and saveN <= 4 then
        local cam = camera()
        if cam then saved[saveN] = cam.CFrame; slots[saveN].save.Text = "SAVED " .. saveN; statusText("Position " .. saveN .. " saved") end
    elseif goN and goN >= 1 and goN <= 4 then
        if saved[goN] then moveToSaved(saved[goN]) else statusText("Save position " .. goN .. " first") end
    end
end

local function registerCommand(name, alias)
    local cmd = Instance.new("TextChatCommand")
    cmd.Name = name
    cmd.PrimaryAlias = alias
    cmd.Parent = TextChatService
    cmd.Triggered:Connect(function(_, text)
        runCommand(text)
    end)
end

RunService:BindToRenderStep(renderName, Enum.RenderPriority.Camera.Value + 1, function(dt)
    local cam = camera()
    if not cam then return end
    if recording and os.clock() - lastSample >= recordInterval then
        lastSample = os.clock()
        captureFrame(os.clock())
    end
    if shakeEnabled and shakeAmount > 0 then
        shakeClock += dt * shakeSpeed
        local strength = shakeAmount * 0.22
        local x = math.noise(shakeClock, 0, 0) * strength
        local y = math.noise(0, shakeClock, 0) * strength
        local roll = math.noise(0, 0, shakeClock) * strength * 0.045
        local pitch = math.noise(shakeClock, 4, 0) * strength * 0.035
        local yaw = math.noise(0, shakeClock, 4) * strength * 0.035
        cam.CFrame = cam.CFrame * CFrame.new(x, y, 0) * CFrame.Angles(pitch, yaw, roll)
    end
end)

script.Destroying:Connect(function()
    recording = false
    cutsceneToken += 1
    pcall(function() RunService:UnbindFromRenderStep(renderName) end)
    if gui then gui:Destroy() end
    -- Disable effects created by this script so they don't linger.
    for _, effect in pairs(post) do
        if effect and effect.Parent then effect.Enabled = false end
    end
end)
